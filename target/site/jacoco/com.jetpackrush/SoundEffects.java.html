package com.jetpackrush;

import java.util.Map;

/** Observateur qui joue le son associé à chaque {@link GameEvent}. */
public class SoundEffects implements GameEventListener {

    private static final Map<GameEvent, String> SOUNDS = Map.of(
        GameEvent.COIN_COLLECTED, "coin_pickup_1.wav",
        GameEvent.PROTECTION_LOST, "dud_zapper_pop.wav",
        GameEvent.PLAYER_SHOCKED, "shock.wav");

    private final SoundPlayer player;

    /**
     * @param player ce qui joue réellement les sons
     */
    public SoundEffects(SoundPlayer player) {
        this.player = player;
    }

    @Override
    public void onEvent(GameEvent event) {
        String sound = SOUNDS.get(event);
        if (sound != null) {
            player.play(sound);
        }
    }

    static String soundFor(GameEvent event) {
        return SOUNDS.get(event);
    }
}
