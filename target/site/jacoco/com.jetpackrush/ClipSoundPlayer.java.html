package com.jetpackrush;

import java.io.BufferedInputStream;
import java.io.IOException;
import java.io.InputStream;
import java.util.HashMap;
import java.util.Map;
import javax.sound.sampled.AudioFormat;
import javax.sound.sampled.AudioInputStream;
import javax.sound.sampled.AudioSystem;
import javax.sound.sampled.Clip;
import javax.sound.sampled.LineUnavailableException;
import javax.sound.sampled.UnsupportedAudioFileException;

/**
 * {@link SoundPlayer} basé sur {@code javax.sound} (fourni avec Java) : chaque son est chargé en
 * mémoire à sa première lecture. Sans carte son ou si le fichier est illisible, le son est
 * simplement ignoré : le jeu continue en silence.
 */
public class ClipSoundPlayer implements SoundPlayer {

    private final Map<String, Clip> clips = new HashMap<>();
    private boolean reported;

    @Override
    public void play(String name) {
        try {
            Clip clip = clips.get(name);
            if (clip == null) {
                clip = load(name);
                clips.put(name, clip);
            }
            clip.stop();
            clip.setFramePosition(0);
            clip.start();
        } catch (IOException | UnsupportedAudioFileException | LineUnavailableException
                 | IllegalArgumentException | IllegalStateException e) {
            if (!reported) {
                reported = true;
                System.err.println("Son indisponible (" + name + ") : " + e);
            }
        }
    }

    private static Clip load(String name)
            throws IOException, UnsupportedAudioFileException, LineUnavailableException {
        try (InputStream in = new BufferedInputStream(Assets.open(name));
             AudioInputStream source = AudioSystem.getAudioInputStream(in)) {
            // Certains .wav sont en nombres flottants : on les convertit en PCM 16 bits, lu partout
            AudioFormat f = source.getFormat();
            AudioFormat pcm = new AudioFormat(AudioFormat.Encoding.PCM_SIGNED, f.getSampleRate(), 16,
                f.getChannels(), f.getChannels() * 2, f.getSampleRate(), false);
            try (AudioInputStream converted = AudioSystem.getAudioInputStream(pcm, source)) {
                byte[] data = converted.readAllBytes();
                Clip clip = AudioSystem.getClip();
                clip.open(pcm, data, 0, data.length);
                return clip;
            }
        }
    }
}
